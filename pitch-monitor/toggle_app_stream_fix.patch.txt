    if session.get('authed'):
        return True, None

    if not supplied:
        # Try JSON body (for POSTs)
        try:
            j = request.get_json(silent=True) or {}

def _normalize_queued_file_for_tenant(wav: str, tenant: str) -> str:
    # Accept either tenant-relative paths like '2026-03-17/.../X.mp3'
    # or absolute-ish UI paths like 'recordings/<tenant>/2026-03-17/.../X.mp3'.
    w = (wav or '').strip().lstrip('/')
    t = (tenant or '').strip().lower()
    if not w or not t:
        return w
    prefix = f"recordings/{t}/"
    if w.lower().startswith(prefix):
        return w[len(prefix):]
    return w

def _http_post_json(url: str, payload: dict | None = None, timeout: float = 2.0) -> tuple[int, str]:
    # stdlib only (avoid new deps)
    import urllib.request
    import urllib.error
    data = None
    headers = {}
    if payload is not None:
        data = json.dumps(payload).encode('utf-8')
        headers['Content-Type'] = 'application/json'
    req = urllib.request.Request(url, data=data, headers=headers, method='POST')
    try:
        with urllib.request.urlopen(req, timeout=timeout) as resp:
            return resp.getcode(), resp.read().decode('utf-8', errors='replace')
    except urllib.error.HTTPError as e:
        return e.code, e.read().decode('utf-8', errors='replace')
    except Exception as e:
        return 0, str(e)

def _aws(args: list[str], timeout: float = 10.0) -> tuple[int, str]:
    # Run AWS CLI and return (rc, combined_output).
    cmd = [LIBRARY_AWS_CLI, *args, "--region", LIBRARY_AWS_REGION]

def _lib_prefix(area: str, subprefix: str = "") -> str:
    area = (area or "").strip().lower()
    if area not in ("recordings", "tracks"):
        area = "recordings"
    tenant = _tenant_from_request(request.host)
    base = f"vps/{LIBRARY_VPS_ID}/{area}/{tenant}/"
    sub = (subprefix or "").lstrip("/")
    sub = sub.replace("..", "")
    if sub and not sub.endswith("/"):
        sub += "/"
    return base + sub


def _safe_library_path_subpath(path_value: str) -> tuple[str, str] | tuple[None, None]:
    """Return (area, subpath) for a user path like recordings/foo/bar/.

    area in {recordings, tracks}; subpath is normalized relative path with trailing slash allowed.
    """
    raw = (path_value or '').strip().lstrip('/')
    if not raw:
        return None, None
    raw = raw.replace('\\', '/')
    if '..' in raw:
        return None, None
    parts = [p for p in raw.split('/') if p]
    if not parts:
        return None, None
    area = parts[0].lower()
    if area not in ('recordings', 'tracks'):
        return None, None
    sub = '/'.join(parts[1:])
    if sub and not sub.endswith('/'):
        sub += '/'
    return area, sub


def _zip_filename_for_subpath(subpath: str, fallback: str = 'session') -> str:
    bits = [b for b in (subpath or '').split('/') if b]
    base = bits[-1] if bits else fallback
    safe = ''.join(ch if ch.isalnum() or ch in ('-','_') else '_' for ch in base)
    return f"{safe or fallback}.zip"

def _list_archived_dirs(area: str) -> set[str]:
    # Return set of quoted dir names that are archived at the area root.
    try:
        s3 = _s3_client()
        pref = _archived_marker_prefix(area)
        out = set()
        token = None
        while True:
            kw = {'Bucket': LIBRARY_S3_BUCKET, 'Prefix': pref}
            if token:
                kw['ContinuationToken'] = token
            resp = s3.list_objects_v2(**kw)
            for obj in resp.get('Contents') or []:
                key = obj.get('Key') or ''
                if not key.startswith(pref):
                    continue
                name = key[len(pref):]
                if name.endswith('.marker'):
                    out.add(name[:-7])
            if resp.get('IsTruncated'):
                token = resp.get('NextContinuationToken')
            else:
                break
        return out
    except Exception:
        return set()
def _s3_list(prefix: str) -> dict:
    # List S3 objects under a prefix.
    import json
    rc1, out1 = _aws([
        "s3api", "list-objects-v2",
        "--bucket", LIBRARY_S3_BUCKET,
        "--prefix", prefix,
        "--delimiter", "/",
    ], timeout=15.0)
    if rc1 != 0:
        return {"ok": False, "error": out1.strip(), "prefix": prefix}

    data = json.loads(out1 or "{}")
    dirs = [cp.get("Prefix") for cp in (data.get("CommonPrefixes") or []) if cp.get("Prefix")]

    files = []
    for obj in (data.get("Contents") or []):
        key = obj.get("Key")
        if (not key) or key.endswith("/") or key == prefix:
            continue
        files.append({
            "key": key,
            "name": key.split("/")[-1],
            "size": obj.get("Size", 0),
            "lastModified": obj.get("LastModified"),
        })

    return {
        "ok": True,
        "prefix": prefix,
        "dirs": [d[len(prefix):].rstrip("/") for d in dirs if d.startswith(prefix)],
        "files": files,
    }

def _s3_presign(key: str, expires: int = 900, attachment: bool = True) -> dict:
    key = (key or "").lstrip("/")
    tenant = _tenant_from_request(request.host)
    allowed1 = f"vps/{LIBRARY_VPS_ID}/recordings/{tenant}/"
    allowed2 = f"vps/{LIBRARY_VPS_ID}/tracks/{tenant}/"
    if not (key.startswith(allowed1) or key.startswith(allowed2)):
        return {"ok": False, "error": "key not allowed"}

    try:
        s3 = _s3_client()
        filename = key.split('/')[-1]
        params = {
            'Bucket': LIBRARY_S3_BUCKET,
            'Key': key,
        }
        if attachment:
            params['ResponseContentDisposition'] = f'attachment; filename="{filename}"'
        else:
            params['ResponseContentDisposition'] = f'inline; filename="{filename}"'
        url = s3.generate_presigned_url(
            ClientMethod='get_object',
            Params=params,
            ExpiresIn=int(expires),
        )
        return {"ok": True, "url": url, "key": key}
    except Exception as e:
        return {"ok": False, "error": str(e)}


# ---------- HELPER: Jamulus PID ----------
def get_jamulus_pid(unit: str = "jamulus-headless.service"):
    """Return Jamulus server PID (systemd first; fallback to pgrep)."""
    try:
        outp = subprocess.run(["systemctl", "show", unit, "-p", "MainPID", "--value"], capture_output=True, text=True).stdout.strip()
        if outp and outp.isdigit() and int(outp) > 0:
            return int(outp)
    except Exception:
        pass
    try:
        outp = subprocess.run(["pgrep", "-f", "/usr/bin/Jamulus -s"], capture_output=True, text=True).stdout.strip()
        return int(outp.splitlines()[0]) if outp else None
    except Exception:
        return None

def _read_jsonrpc_secret() -> str | None:
    try:
        with open(JSONRPC_SECRET_FILE, "r") as f:
            return f.read().strip()
    except Exception:
        return None


def _jsonrpc_send_lines(lines: list[str], timeout: float = 2.0, port: int | None = None) -> list[dict]:
    """Send newline-delimited JSON-RPC messages over TCP; returns decoded JSON responses."""
    data = ("\n".join(lines) + "\n").encode("utf-8")
    out: list[dict] = []
    with socket.create_connection((JSONRPC_HOST, port or JSONRPC_PORT), timeout=timeout) as s:
        s.settimeout(timeout)
        s.sendall(data)
        buf = b""
        # Read up to len(lines) responses (auth + request)
        while len(out) < len(lines):
            chunk = s.recv(65536)
            if not chunk:
                break
            buf += chunk
            while b"\n" in buf and len(out) < len(lines):
                raw, buf = buf.split(b"\n", 1)
                raw = raw.strip()
                if not raw:
                    continue
                out.append(json.loads(raw.decode("utf-8")))
    return out


def jamulus_rpc_request(method: str, params: dict | None = None, tenant: str | None = None) -> dict:
    secret = _read_jsonrpc_secret()

    port = _tenant_rpc_port(tenant) if tenant else JSONRPC_PORT
    if not secret:
        raise RuntimeError("JSON-RPC secret file missing/unreadable")

    auth = json.dumps({"id": 1, "jsonrpc": "2.0", "method": "jamulus/apiAuth", "params": {"secret": secret}})
    req = json.dumps({"id": 2, "jsonrpc": "2.0", "method": method, "params": params or {}})
    resps = _jsonrpc_send_lines([auth, req], port=port)
    if len(resps) < 2:
        raise RuntimeError("JSON-RPC: incomplete response")

    if resps[0].get("result") != "ok":
        raise RuntimeError(f"JSON-RPC auth failed: {resps[0]}")

    if "error" in resps[1]:
        raise RuntimeError(f"JSON-RPC error: {resps[1]['error']}")

    return resps[1]

def _has_recent_recording_writes(active_window_seconds: int = 15, recordings_dir: str | None = None) -> bool:
    """Fallback heuristic: consider recording active if files were written recently in newest Jam-* dir."""
    try:
        rd = recordings_dir or RECORDINGS_DIR
        if not os.path.isdir(rd):
            return False
        jam_dirs = sorted(
            (d for d in glob.glob(os.path.join(rd, "Jam-*")) if os.path.isdir(d)),
            key=lambda p: os.path.getmtime(p),
            reverse=True,
        )
        if not jam_dirs:
            return False
        newest = jam_dirs[0]
        newest_mtime = 0.0
        for root, _dirs, files in os.walk(newest):
            for fn in files:
                try:
                    mt = os.path.getmtime(os.path.join(root, fn))
                    newest_mtime = max(newest_mtime, mt)
                except FileNotFoundError:
                    pass
        if newest_mtime == 0.0:
            return False
        return (datetime.now().timestamp() - newest_mtime) <= active_window_seconds
    except Exception:
        return False




def _has_unnamed_client() -> bool:
    """Detect unnamed clients from Jamulus status HTML list."""
    try:
        t = Path('/var/lib/jamulus/status/jamulus_status.html').read_text(encoding='utf-8', errors='ignore')
        return '<li></li>' in t or '<li> </li>' in t
    except Exception:
        return False


def _preflight_clear_unnamed_idle() -> bool:
    """If unnamed client persists beyond threshold, reset Jamulus once."""
    try:
        if not _has_unnamed_client():
            try:
                Path(UNNAMED_SEEN_FILE).unlink(missing_ok=True)
            except Exception:
                pass
            return False

        now = time.time()
        seen_path = Path(UNNAMED_SEEN_FILE)
        first_seen = None
        if seen_path.exists():
            try:
                first_seen = float(seen_path.read_text().strip())
            except Exception:
                first_seen = None
        if first_seen is None:
            seen_path.write_text(str(now))
            return False

        if (now - first_seen) < UNNAMED_IDLE_RESET_SECONDS:
            return False

        if _has_recent_recording_writes(active_window_seconds=15):
            return False

        r = subprocess.run(["sudo", "-n", "systemctl", "restart", "jamulus-headless.service"], capture_output=True, text=True)
        if r.returncode == 0:
            seen_path.write_text(str(time.time()))
            time.sleep(2)
            return True
        return False
    except Exception:
        return False

@app.route("/toggle-recording", methods=["POST"])
def toggle_recording():
    ok, resp = _require_passcode()
    if not ok: return resp
    data  = request.get_json(force=True)
    name  = (data.get("session_name") or "").strip()
    action = (data.get("action") or "").strip().lower()
    include_injector = bool(data.get("include_injector", False))

    # Persist user preference for uploader behavior
    try:
        Path(INCLUDE_INJECTOR_FLAG).write_text("1" if include_injector else "0")
    except Exception:
        pass
    try:
        with open(INCLUDE_INJECTOR_MAP_CSV, "a", newline="") as f:
            f.write(f"{_recording_map_key_now()},{1 if include_injector else 0}\n")
    except Exception:
        pass

    if action not in ("start", "stop"):
        return "Invalid action", 400

    tenant=_tenant_id_from_host()
    unit=_tenant_jamulus_unit(tenant)
    pid = get_jamulus_pid(unit)
    if not pid:
        return "Jamulus server not running", 500

    # Prefer JSON-RPC explicit start/stop + true enabled/disabled state.
    # If JSON-RPC isn't available yet, fall back to the file-write heuristic.
    is_enabled = jamulus_recording_enabled(tenant=tenant)
    if is_enabled is None:
        is_enabled = _has_recent_recording_writes(recordings_dir=_tenant_recordings_dir(tenant))

    desired_active = (action == "start")

    # Pre-flight: if starting and unnamed idle clients persist, clear them automatically.
    if action == "start":
        _preflight_clear_unnamed_idle()
        pid = get_jamulus_pid(unit)
        if not pid:
            return "Jamulus server not running after preflight reset", 500

    # Pre-flight: if starting and a name is provided, enforce duplicate-name rule BEFORE starting recording
    if action == "start" and name:
        if name_used_recently(name, HOURS_BLOCK):
            msg = f"⏱ Name '{name}' was used in the last {HOURS_BLOCK} h"
            print(msg)
            return msg, 400

    if bool(is_enabled) == desired_active:
        # Already in the desired state; avoid accidentally toggling the wrong way.
        if action == "start" and name:
            # Still allow naming/logging without toggling.
            if name_used_recently(name, HOURS_BLOCK):
                msg = f"⏱ Name '{name}' was used in the last {HOURS_BLOCK} h"
                print(msg)
                return msg, 400
            save_session_name(name)
            log_session_name(name)
            with open(RECORDING_MAP_CSV, "a") as f:
                f.write(f"{_recording_map_key_now()},{name}\n")
        return "OK", 200

    # Trigger Jamulus.
    if jamulus_recording_enabled(tenant=tenant) is not None:
        try:
            method = "jamulusserver/startRecording" if action == "start" else "jamulusserver/stopRecording"
            jamulus_rpc_request(method, tenant=tenant)
        except Exception as e:
            return f"Failed to {action} recording via JSON-RPC: {e}", 500
    else:
        # Fallback: toggle
        res = _jamulus_toggle_recording(pid)
        if res.returncode != 0:
            msg = (res.stderr or res.stdout or "").strip()
            return f"Failed to toggle Jamulus recording (sudo/kill). {msg}", 500

    if action == "start":
        if name:
            if name_used_recently(name, HOURS_BLOCK):
                msg = f"⏱ Name '{name}' was used in the last {HOURS_BLOCK} h"
                print(msg)
                return msg, 400
            save_session_name(name)
            log_session_name(name)
            with open(RECORDING_MAP_CSV, "a") as f:
                f.write(f"{_recording_map_key_now()},{name}\n")
        Path(RECORDING_FLAG).write_text("ON")
    else:  # stop
        if os.path.exists(RECORDING_FLAG):
            os.remove(RECORDING_FLAG)
    return "OK"

def _support_status_payload() -> dict:
    enabled = jamulus_recording_enabled()
    if enabled is None:
        enabled = _has_recent_recording_writes()

    jamulus_running = False
    try:
        result = subprocess.run(['pgrep', '-x', 'jamulus'], capture_output=True)
        jamulus_running = result.returncode == 0
    except Exception:
        pass

    # Lightweight server-side quality indicators (infrastructure health only)
    try:
        l1, l5, l15 = os.getloadavg()
    except Exception:
        l1, l5, l15 = (0.0, 0.0, 0.0)
    cores = os.cpu_count() or 1
    load_ratio = float(l1) / float(max(1, cores))

    if not jamulus_running:
        grade = 'red'
        label = 'Server issue'
    elif load_ratio < 0.70:
        grade = 'green'
        label = 'Good'
    elif load_ratio < 1.10:
        grade = 'yellow'
        label = 'Busy'
    else:
        grade = 'red'
        label = 'Overloaded'

    quality = {
        'grade': grade,
        'label': label,
        'load1': round(float(l1), 2),
        'load5': round(float(l5), 2),
        'cores': int(cores),
        'load_ratio': round(load_ratio, 2),
        'note': 'Server-side health only; user network/device quality may still vary.'
    }

    return {
        'ok': True,
        'server_id': LIBRARY_VPS_ID,
        'jamulus_running': jamulus_running,
        'recording_state': 'ON' if enabled else 'OFF',
        'status_text': 'Online' if jamulus_running else 'Offline',
        'quality': quality,
    }


def _support_reply(intent: str, text: str = '') -> dict:
    intent = (intent or '').strip().lower()
    text_l = (text or '').strip().lower()
    if not intent:
        if 'restart' in text_l:
            intent = 'restart'
        elif 'status' in text_l or 'availability' in text_l or 'up' in text_l:
            intent = 'status'
        elif 'audio' in text_l or 'hear' in text_l or 'latency' in text_l or 'connect' in text_l:
            intent = 'troubleshoot'
        elif 'human' in text_l or 'escalate' in text_l or 'help me' in text_l:
            intent = 'escalate'
        else:
            intent = 'help'

    if intent in ('status', 'availability'):
        st = _support_status_payload()
        return {
            'ok': True,
            'intent': 'status',
            'message': f"Server {st['server_id']} is {st['status_text']}. Recording is {st['recording_state']}.",
            'data': st,
            'quick_actions': ['restart', 'troubleshoot', 'escalate']
        }

    if intent == 'restart':
        url = f"{TRACKBOT_BASE_URL}/api/jamulus/restart"
        code, body = _http_get(url, timeout=30.0)
        try:
            data = json.loads(body) if body else {'ok': False}
        except Exception:
            data = {'ok': False, 'error': body}
        ok = bool(code and 200 <= code < 300 and data.get('ok') is True)
        msg = 'Restart requested successfully. Please retry in 20-30 seconds.' if ok else 'Restart failed. I can escalate this to a human now.'
        return {'ok': ok, 'intent': 'restart', 'message': msg, 'data': {'upstream_code': code, 'upstream': data}, 'quick_actions': ['status', 'escalate']}

    if intent == 'troubleshoot':
        steps = [
            '1) Confirm Jamulus server address/port are correct.',
            '2) Reconnect and verify your nickname/input device.',
            '3) If audio stutters, increase buffer/latency slightly.',
            '4) If still failing, use Restart and rejoin after 30 seconds.'
        ]
        return {'ok': True, 'intent': 'troubleshoot', 'message': 'Try this quick checklist:\n' + '\n'.join(steps), 'quick_actions': ['status', 'restart', 'escalate']}

    if intent == 'escalate':
        return {'ok': True, 'intent': 'escalate', 'message': 'Escalation requested. A human operator will review this server issue shortly.', 'quick_actions': ['status']}

    return {
        'ok': True,
        'intent': 'help',
        'message': 'I can help with: status, availability, restart, troubleshooting, or escalation.',
        'quick_actions': ['status', 'restart', 'troubleshoot', 'escalate']
    }

def _current_jack_period() -> int | None:
    try:
        r = subprocess.run(["ps", "-eo", "cmd"], capture_output=True, text=True)
        txt = (r.stdout or "")
        for line in txt.splitlines():
            if "/usr/bin/jackd -r -d dummy" in line:
                m = re.search(r"\s-p\s+(\d+)", line)
                if m:
                    return int(m.group(1))
    except Exception:
        pass
    try:
        cfg = Path('/etc/systemd/system/jackd.service.d/override.conf')
        if cfg.exists():
            t = cfg.read_text()
            m = re.search(r"\s-p\s+(\d+)", t)
            if m:
                return int(m.group(1))
    except Exception:
        pass
    return None

    area = 'tracks' if sub.startswith('tracks/') else 'recordings'

    if area == 'tracks':
        rel = sub[len('tracks/'):].lstrip('/') if sub.startswith('tracks/') else sub

                try:
                    if isinstance(dirs, list):
                        dirs = [d for d in dirs if str(d) not in ('trash',)]
                except Exception:
                    pass

                return jsonify({"ok": False, "error": "cross-tenant tracks browse blocked"}), 403
            if first == tenant:
                rel = rel.split('/', 1)[1] if '/' in rel else ''
        prefix = f"vps/{LIBRARY_VPS_ID}/tracks/{tenant}/" + (rel or '')
    else:
        rel = sub[len('recordings/'):].lstrip('/') if sub.startswith('recordings/') else sub
        if rel and '/' in rel:
            first = rel.split('/', 1)[0]
            if first in ('pd','vc','seigr') and first != tenant:
                return jsonify({"ok": False, "error": "cross-tenant recordings browse blocked"}), 403
            if first == tenant:
                rel = rel.split('/', 1)[1] if '/' in rel else ''
        prefix = f"vps/{LIBRARY_VPS_ID}/recordings/{tenant}/" + (rel or '')

    try:
        result = _s3_list(prefix)
        # JB_HIDE_TRASH_FILTER: never show trash/ marker folders in UI listings
        try:
            _dirs = result.get('dirs', [])
            if isinstance(_dirs, list):
                _dirs = [d for d in _dirs if str(d) not in ('trash',)]
            result['dirs'] = _dirs
        except Exception:
            pass
        # Ensure per-tenant temp folder is always visible at recordings root
        try:
            if area == 'recordings' and (rel or '') == '' :
                if isinstance(result.get('dirs', None), list) and 'temp' not in result['dirs']:
                    result['dirs'] = ['temp'] + result['dirs']
        except Exception:
            pass
        return jsonify({"ok": True, "dirs": result.get('dirs', []), "files": result.get('files', []), "prefix": prefix})
    except Exception as e:
        return jsonify({"ok": False, "error": str(e)}), 500

@app.route('/wav/delete', methods=['POST'])
def wav_delete():
    """Delete a file or folder (prefix) inside the tenant-scoped library.

    Safety:
    - Only allows deletes within:
      - vps/<LIBRARY_VPS_ID>/recordings/<tenant>/...
      - vps/<LIBRARY_VPS_ID>/tracks/<tenant>/...
      - vps/<LIBRARY_VPS_ID>/recordings/<tenant>/temp/... (per-tenant scratch)
    """
    ok, resp = _require_passcode()
    if not ok:
        return resp

    tenant = _tenant_from_request(request.host)
    d = request.get_json(silent=True) or {}
    rel_path = (d.get('path') or d.get('file') or '').strip().lstrip('/')
    kind = (d.get('kind') or '').strip().lower()  # 'file' | 'dir'

    if not rel_path:
        return jsonify({'ok': False, 'error': 'missing path'}), 400

    # Normalize kind
    if kind not in ('file','dir'):
        # infer from trailing slash
        kind = 'dir' if rel_path.endswith('/') else 'file'

    # Reuse the same tenant isolation rules as wav_browse, but produce an S3 prefix/key.
    # rel_path is like: recordings/<tenant>/YYYY-MM-DD/... OR tracks/<tenant>/...
    sub = rel_path
    area = 'tracks' if sub.startswith('tracks/') else 'recordings'

    if area == 'tracks':
        rel = sub[len('tracks/'):].lstrip('/') if sub.startswith('tracks/') else sub
        if rel and '/' in rel:
            first = rel.split('/', 1)[0]
            if first in ('pd','vc','seigr') and first != tenant:
                return jsonify({'ok': False, 'error': 'cross-tenant tracks delete blocked'}), 403
            if first == tenant:
                rel = rel.split('/', 1)[1] if '/' in rel else ''
        prefix = f"vps/{LIBRARY_VPS_ID}/tracks/{tenant}/" + (rel or '')
    else:
        rel = sub[len('recordings/'):].lstrip('/') if sub.startswith('recordings/') else sub
        if rel.startswith('temp/'):
            prefix = f"vps/{LIBRARY_VPS_ID}/recordings/" + rel
        else:
            if rel and '/' in rel:
                first = rel.split('/', 1)[0]
                if first in ('pd','vc','seigr') and first != tenant:
                    return jsonify({'ok': False, 'error': 'cross-tenant recordings delete blocked'}), 403
                if first == tenant:
                    rel = rel.split('/', 1)[1] if '/' in rel else ''
            prefix = f"vps/{LIBRARY_VPS_ID}/recordings/{tenant}/" + (rel or '')

    # For file delete, prefix is the full key.
    # Instead of hard-delete, we move to a tenant-scoped trash prefix that expires via lifecycle.
    try:
        s3 = _s3_client()
        deleted = 0

        # Trash root: vps/<VPS>/<area>/<tenant>/trash/<stamp>/...
        stamp = _recording_map_key_now()

        def _trash_dest_for_key(area: str, tenant: str, key: str) -> str:
            base = f"vps/{LIBRARY_VPS_ID}/{area}/{tenant}/"
            if key.startswith(base):
                rel = key[len(base):]
                return f"{base}trash/{stamp}/" + rel
            # recordings/temp/... goes to recordings/temp/trash/<tenant>/<stamp>/...
            temp_base = f"vps/{LIBRARY_VPS_ID}/recordings/{tenant}/temp/"
            if key.startswith(temp_base):
                rel = key[len(temp_base):]
                return f"{temp_base}trash/{stamp}/" + rel
            # fallback (should not happen given prefix guards)
            return f"vps/{LIBRARY_VPS_ID}/recordings/{tenant}/temp/trash/{stamp}/" + key.replace('/', '_')

        def _move_key_to_trash(key: str) -> str:
            dest = _trash_dest_for_key(area, tenant, key)
            # Copy then delete (S3 has no rename)
            s3.copy_object(
                Bucket=LIBRARY_S3_BUCKET,
                Key=dest,
                CopySource={'Bucket': LIBRARY_S3_BUCKET, 'Key': key},
            )
            s3.delete_object(Bucket=LIBRARY_S3_BUCKET, Key=key)
            return dest

        if kind == 'file':
            key = prefix
            dest = _move_key_to_trash(key)
            deleted = 1
            return jsonify({'ok': True, 'deleted': deleted, 'kind': 'file', 'key': key, 'trash_key': dest})

        # Directory/prefix delete (move all objects under prefix to trash)
        pref = prefix
        if not pref.endswith('/'):
            pref = pref + '/'

        keys = []
        token = None
        while True:
            kw = {'Bucket': LIBRARY_S3_BUCKET, 'Prefix': pref}
            if token:
                kw['ContinuationToken'] = token
            resp = s3.list_objects_v2(**kw)
            for obj in (resp.get('Contents') or []):
                k = obj.get('Key') or ''
                if k and not k.endswith('/'):
                    keys.append(k)
            if resp.get('IsTruncated'):
                token = resp.get('NextContinuationToken')
            else:
                break

        # Nothing to delete is still "ok".
        if not keys:
            return jsonify({'ok': True, 'deleted': 0, 'kind': 'dir', 'prefix': pref, 'note': 'no objects under prefix'})

        moved = 0
        last_dest = None
        for k in keys:
            last_dest = _move_key_to_trash(k)
            moved += 1

        return jsonify({'ok': True, 'deleted': moved, 'kind': 'dir', 'prefix': pref, 'trash_example': last_dest})
    except Exception as e:
        return jsonify({'ok': False, 'error': str(e)}), 500

@app.route("/wav/download-folder")
def wav_download_folder():
    ok, resp = _require_passcode()
    if not ok:
        return resp

    area, sub = _safe_library_path_subpath(request.args.get('path') or '')

    # JB_STRIP_TENANT_FROM_FOLDER_PATH: accept paths that redundantly include tenant (recordings/<tenant>/...)
    tenant = _tenant_from_request(request.host)
    if sub and sub.startswith(tenant + '/'):  # tolerate caller-provided tenant
        sub = sub[len(tenant)+1:]
    if not area:
        return jsonify({"ok": False, "error": "invalid or missing folder path"}), 400

    prefix = _lib_prefix(area, sub)
    try:
        s3 = _s3_client()
        token = None
        keys = []
        while True:
            kw = {'Bucket': LIBRARY_S3_BUCKET, 'Prefix': prefix}
            if token:
                kw['ContinuationToken'] = token
            resp = s3.list_objects_v2(**kw)
            for obj in resp.get('Contents') or []:
                key = obj.get('Key') or ''
                if not key or key.endswith('/'):
                    continue
                if key.startswith(prefix):
                    keys.append(key)
            if resp.get('IsTruncated'):
                token = resp.get('NextContinuationToken')
            else:
                break

        if not keys:
            return jsonify({"ok": False, "error": "folder is empty"}), 404

        tmp = tempfile.NamedTemporaryFile(prefix='jamulus_session_', suffix='.zip', delete=False)
        tmp_path = tmp.name
        tmp.close()

        with zipfile.ZipFile(tmp_path, 'w', compression=zipfile.ZIP_DEFLATED) as zf:
            for key in keys:
                rel = key[len(prefix):]
                if not rel:
                    continue
                body = s3.get_object(Bucket=LIBRARY_S3_BUCKET, Key=key)['Body'].read()
                zf.writestr(rel, body)

        dl_name = _zip_filename_for_subpath(sub or area, fallback='session')
        response = send_file(tmp_path, as_attachment=True, download_name=dl_name, mimetype='application/zip')

        @response.call_on_close
        def _cleanup_tmp_zip():
            try:
                os.remove(tmp_path)
            except Exception:
                pass

        return response
    except Exception as e:
        return jsonify({"ok": False, "error": str(e)}), 500

@app.route('/api/library/presign-put')
def api_library_presign_put():
    ok, resp = _require_passcode()
    if not ok: return resp

    area = (request.args.get('area') or 'tracks').strip().lower()
    prefix = (request.args.get('prefix') or '').strip()
    filename = (request.args.get('filename') or '').strip()
    content_type = (request.args.get('contentType') or 'application/octet-stream').strip()

    if not filename:
        return jsonify({'ok': False, 'error': 'missing filename'}), 400

    # basic filename sanitization
    filename = filename.replace('..', '').replace('\\', '/').split('/')[-1]
    if not filename:
        return jsonify({'ok': False, 'error': 'bad filename'}), 400

    # allowlist extensions
    low = filename.lower()
    if not (low.endswith('.mp3') or low.endswith('.wav') or low.endswith('.m4a') or low.endswith('.flac')):
        return jsonify({'ok': False, 'error': 'file type not allowed'}), 400

    # Prefer uploading to tracks by default; recordings can be enabled but keep it explicit
    if area not in ('tracks','recordings'):
        area = 'tracks'

    key = _lib_prefix(area, prefix) + filename

    # only allow within vps/<id>/(tracks|recordings)/
    tenant = _tenant_from_request(request.host)
    allowed1 = f"vps/{LIBRARY_VPS_ID}/recordings/{tenant}/"
    allowed2 = f"vps/{LIBRARY_VPS_ID}/tracks/{tenant}/"
    if not (key.startswith(allowed1) or key.startswith(allowed2)):
        return jsonify({'ok': False, 'error': 'key not allowed'}), 403

    expires = int(request.args.get('expires') or '900')
    expires = max(60, min(expires, 3600))

    try:
        s3 = _s3_client()
        url = s3.generate_presigned_url(
            ClientMethod='put_object',
            Params={
                'Bucket': LIBRARY_S3_BUCKET,
                'Key': key,
                'ContentType': content_type,
            },
            ExpiresIn=int(expires),
        )
        return jsonify({'ok': True, 'url': url, 'key': key})
    except Exception as e:
        return jsonify({'ok': False, 'error': str(e)}), 500

def _load_names():
    if os.path.exists(SAVED_NAMES_CSV):
        with open(SAVED_NAMES_CSV) as f:

    return set()

    with open(SAVED_NAMES_CSV, "w") as f:
        for n in sorted(names, key=str.lower):
            f.write(f"{n}\n")

def name_used_recently(name, hours=10):
    if not os.path.exists(SESSION_LOG_JSN):
        return False
    try:
        with open(SESSION_LOG_JSN, "r") as f:
            log = json.load(f)
    except Exception as e:
        print(f"[LOG] JSON parse failed: {e}")
        return False

    now = _now_local()
    cutoff = now - timedelta(hours=hours)

    print(f"[LOG] Checking for recent use of '{name}' — now: {now}, cutoff: {cutoff}")

    for entry in log:
        if entry.get("name") == name:
            try:
                timestamp = datetime.fromisoformat(entry.get("timestamp"))
                # Older entries may be naive; assume local time.
                if timestamp.tzinfo is None and _LOCAL_TZ is not None:
                    timestamp = timestamp.replace(tzinfo=_LOCAL_TZ)
                print(f"[LOG] Found '{name}' entry at {timestamp}")
                if timestamp > cutoff:
                    print(f"[LOG] Name '{name}' used recently!")
                    return True
            except Exception as e:
                print(f"[LOG] Timestamp parse failed: {e}")
                continue
    return False

def log_session_name(name):
